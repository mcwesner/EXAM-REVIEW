#include <Arduino.h>

#define BUTTON_PIN 21
#define DEBOUNCE_MS 100

int previousRaw = HIGH;
int stableButton = HIGH;
uint32_t lastRawChange = 0;
int mode = 0;

void runModeAction()
{
  if (mode == 0)
  {
    // Change the action for mode 0.
    Serial.println("Mode 0");
  }
  else if (mode == 1)
  {
    // Change the action for mode 1.
    Serial.println("Mode 1");
  }
  else
  {
    // Change the action for mode 2.
    Serial.println("Mode 2");
  }
}

void setup()
{
  Serial.begin(9600);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  previousRaw = digitalRead(BUTTON_PIN);
  lastRawChange = millis();
}

void loop()
{
  uint32_t now = millis();
  int raw = digitalRead(BUTTON_PIN);

  if (raw != previousRaw)
  {
    previousRaw = raw;
    lastRawChange = now;
  }

  if (raw != stableButton &&
      now - lastRawChange >= DEBOUNCE_MS)
  {
    stableButton = raw;

    if (stableButton == LOW)
    {
      mode = (mode + 1) % 3;
      runModeAction();
    }
  }
}
